cmake_minimum_required(VERSION 3.24)
project(ShadowVeil VERSION 0.9.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# SV_RELEASE: a build for players. Content is found next to the executable only (never in the
# source checkout), Windows builds have no console window, and the runtime is linked statically.
option(SV_RELEASE "Build a distributable release" OFF)
# SV_BUNDLED_DEPS: fetch pinned SDL3 / FreeType / zlib and link them statically instead of using
# the system's. Default on for Windows and for release builds (portable Linux tarballs).
if(WIN32 OR SV_RELEASE)
  set(_sv_bundle_default ON)
else()
  set(_sv_bundle_default OFF)
endif()
option(SV_BUNDLED_DEPS "Build pinned SDL3, FreeType and zlib from source" ${_sv_bundle_default})
option(SV_BUILD_TOOLS "Build the test runner and trailer tools" ON)

if(SV_BUNDLED_DEPS)
  include(FetchContent)
  set(FETCHCONTENT_QUIET ON)
  # zlib
  FetchContent_Declare(zlib GIT_REPOSITORY https://github.com/madler/zlib.git GIT_TAG v1.3.2 GIT_SHALLOW TRUE
                       EXCLUDE_FROM_ALL)
  set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
  set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
  # FreeType, without optional codecs (the game ships plain OTF/TTF)
  FetchContent_Declare(freetype GIT_REPOSITORY https://github.com/freetype/freetype.git GIT_TAG VER-2-15-0 GIT_SHALLOW TRUE
                       EXCLUDE_FROM_ALL)
  set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
  # SDL3, static. On Linux it still loads X11 / Wayland / PipeWire / PulseAudio at runtime.
  FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.4.18 GIT_SHALLOW TRUE
                       EXCLUDE_FROM_ALL)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(zlib freetype SDL3)
  if(TARGET zlibstatic)
    set(SV_ZLIB zlibstatic)
  else()
    set(SV_ZLIB ZLIB::ZLIBSTATIC)
  endif()
  set(SV_SDL SDL3::SDL3-static)
  set(SV_FREETYPE freetype)
else()
  find_package(SDL3 REQUIRED CONFIG)
  find_package(Freetype REQUIRED)
  find_package(ZLIB REQUIRED)
  set(SV_ZLIB ZLIB::ZLIB)
  set(SV_SDL SDL3::SDL3)
  set(SV_FREETYPE Freetype::Freetype)
endif()

if(WIN32)
  set(SV_GL opengl32)
else()
  set(OpenGL_GL_PREFERENCE GLVND)
  find_package(OpenGL REQUIRED COMPONENTS OpenGL)
  set(SV_GL OpenGL::OpenGL)
endif()

file(GLOB_RECURSE SV_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM SV_SOURCES ${CMAKE_SOURCE_DIR}/src/main.cpp)

# All game code lives in a library so the test runner can drive the simulation headless.
add_library(sv_engine STATIC ${SV_SOURCES})
target_include_directories(sv_engine PUBLIC src)
target_compile_definitions(sv_engine PUBLIC
  SV_SOURCE_DIR="${CMAKE_SOURCE_DIR}"
  SV_VERSION="${PROJECT_VERSION}")
if(SV_RELEASE)
  target_compile_definitions(sv_engine PUBLIC SV_RELEASE=1)
endif()
if(MSVC)
  target_compile_options(sv_engine PUBLIC /W3 /utf-8 /permissive- /Zc:__cplusplus /wd4244 /wd4267 /wd4305)
  target_compile_definitions(sv_engine PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(sv_engine PUBLIC -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
endif()
if(WIN32)
  target_compile_definitions(sv_engine PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MINGW)
  # C99 printf (%zu, %lld); the default MSVCRT one misreads them
  target_compile_definitions(sv_engine PUBLIC __USE_MINGW_ANSI_STDIO=1)
endif()
target_link_libraries(sv_engine PUBLIC ${SV_SDL} ${SV_GL} ${SV_FREETYPE} ${SV_ZLIB})

# Steam: -DSV_STEAM=ON -DSTEAMWORKS_SDK=/path/to/steamworks_sdk/sdk -DSV_STEAM_APPID=<app id>
option(SV_STEAM "Route achievements to Steam (needs the Steamworks SDK)" OFF)
if(SV_STEAM)
  if(NOT STEAMWORKS_SDK OR NOT SV_STEAM_APPID)
    message(FATAL_ERROR "SV_STEAM needs -DSTEAMWORKS_SDK=<sdk dir> and -DSV_STEAM_APPID=<app id>")
  endif()
  if(WIN32)
    set(SV_STEAM_LIB ${STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.lib)
    set(SV_STEAM_DLL ${STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.dll)
  else()
    set(SV_STEAM_LIB ${STEAMWORKS_SDK}/redistributable_bin/linux64/libsteam_api.so)
    set(SV_STEAM_DLL ${SV_STEAM_LIB})
  endif()
  target_include_directories(sv_engine PUBLIC ${STEAMWORKS_SDK}/public)
  target_compile_definitions(sv_engine PUBLIC SV_STEAM=1 SV_STEAM_APPID=${SV_STEAM_APPID})
  target_link_libraries(sv_engine PUBLIC ${SV_STEAM_LIB})
  install(FILES ${SV_STEAM_DLL} DESTINATION .)
endif()

add_executable(shadow_veil src/main.cpp)
target_link_libraries(shadow_veil PRIVATE sv_engine)
if(WIN32)
  set_target_properties(shadow_veil PROPERTIES OUTPUT_NAME ShadowVeil)
  set(SV_ICO ${CMAKE_SOURCE_DIR}/assets/icon/shadow-veil.ico)
  configure_file(packaging/windows/shadow_veil.rc.in ${CMAKE_BINARY_DIR}/shadow_veil.rc @ONLY)
  target_sources(shadow_veil PRIVATE ${CMAKE_BINARY_DIR}/shadow_veil.rc)
  if(SV_RELEASE)
    set_target_properties(shadow_veil PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  if(MINGW)
    # one self-contained .exe: no libstdc++ / libgcc / winpthread DLLs to ship
    target_link_options(shadow_veil PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(MSVC)
    set_property(TARGET shadow_veil sv_engine PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
elseif(SV_RELEASE)
  target_link_options(shadow_veil PRIVATE -static-libgcc -static-libstdc++)
  if(SV_STEAM)
    set_target_properties(shadow_veil PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endif()
endif()

if(SV_BUILD_TOOLS)
  add_executable(sv_score tools/trailer/score.cpp)
  target_link_libraries(sv_score PRIVATE sv_engine)
  add_executable(sv_tests tests/tests.cpp)
  target_link_libraries(sv_tests PRIVATE sv_engine)
  if(MINGW)
    target_link_options(sv_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  enable_testing()
  add_test(NAME unit_and_simulation COMMAND sv_tests)
endif()

# Install layout.
#   Windows / portable Linux: everything in one folder (ShadowVeil.exe or shadow_veil, data/, assets/).
#   Linux system install:     bin/shadow_veil, share/shadow-veil/{data,assets}, desktop entry and icons.
if(WIN32 OR SV_RELEASE)
  install(TARGETS shadow_veil RUNTIME DESTINATION .)
  install(DIRECTORY data assets DESTINATION . PATTERN "README.md" EXCLUDE PATTERN "*.ico" EXCLUDE)
  install(FILES LICENSE.txt THIRD_PARTY_NOTICES.txt DESTINATION . OPTIONAL)
else()
  install(TARGETS shadow_veil RUNTIME DESTINATION bin)
  install(DIRECTORY data assets DESTINATION share/shadow-veil PATTERN "README.md" EXCLUDE PATTERN "*.ico" EXCLUDE)
  install(FILES packaging/linux/dev.shadowveil.game.desktop DESTINATION share/applications)
  install(FILES packaging/linux/dev.shadowveil.game.metainfo.xml DESTINATION share/metainfo)
  foreach(s 16 32 48 64 128 256 512)
    install(FILES assets/icon/shadow-veil-${s}.png DESTINATION share/icons/hicolor/${s}x${s}/apps RENAME dev.shadowveil.game.png)
  endforeach()
endif()
